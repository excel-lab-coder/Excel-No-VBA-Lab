全角QR生成のVBA完全版コード
― 外部ライブラリなしでBMP生成・セル配置まで行う公開版

この記事では、全角文字を含むQRコードをExcel VBAとWindows標準APIだけで生成する公開版コードを掲載します。閉域・制限環境では、Shift-JISでQRを作り、オプトエレクトロニクス側は漢字出力モード②とShift-JIS設定でExcelなどへ出力する運用を中心に整理します。UTF-8はMOSTが必要です。EdgeなどへShift-JISをそのまま出すと文字化けしやすく、旧IME設定で暫定回避できる場合はありますが、「～」などが「?」になる制限があります。なお、本記事は社内ルールや情報システム部門の管理を回避するためのものではなく、承認済み環境で代替入力手段を検討するための技術メモです。

✅ 結論だけ先に

  • Excelへ出すなら：Shift-JIS＋漢字出力モード②。MOSTなしで運用しやすい設定です。

Private Function ChooseEncoding() As Boolean
    InitEnc

    Dim ans As Long
    Dim msg As String

    msg = "QRコードの文字コードを選択してください。"
    msg = msg & vbCrLf & vbCrLf
    msg = msg & "[はい] = UTF-8(ECI付与)" & vbCrLf

        ChooseEncoding = True
    ElseIf ans = vbNo Then
        gUseUTF8 = False
        ChooseEncoding = True
    Else
        ChooseEncoding = False
    End If
End Function

Private Sub InitGF()
    If gfReady Then Exit Sub

    Dim v As Long
    Dim i As Long

    v = 1

    For i = 0 To 254
        gfEx(i) = v
        gfLg(v) = i

        v = v + v
        If v >= 256 Then v = v Xor 285
    Next i

    For i = 255 To 510
        gfEx(i) = gfEx(i - 255)
    Next i

    gfReady = True
End Sub

Private Function GFMul(ByVal a As Long, ByVal b As Long) As Long
    If a = 0 Or b = 0 Then
        GFMul = 0
        Exit Function
    End If

    GFMul = gfEx((gfLg(a) + gfLg(b)) Mod 255)
End Function

Private Function PowOf2(ByVal n As Long) As Long
    Dim r As Long
    Dim i As Long

    r = 1

    For i = 1 To n
        r = r * 2
    Next i

    PowOf2 = r
End Function

Private Function RSDiv(ByVal degree As Long) As Byte()
    Dim r() As Byte
    Dim i As Long
    Dim j As Long
    Dim rt As Long

    ReDim r(0 To degree - 1) As Byte

    r(degree - 1) = 1
    rt = 1

    For i = 0 To degree - 1
        For j = 0 To degree - 1
            r(j) = CByte(GFMul(CLng(r(j)), rt))

            If j + 1 < degree Then
                r(j) = r(j) Xor r(j + 1)
            End If
        Next j

        rt = GFMul(rt, 2)
    Next i

    RSDiv = r
End Function

Private Function RSRem(data() As Byte, ByVal dLen As Long, dv() As Byte, ByVal dvLen As Long) As Byte()
    Dim r() As Byte
    Dim i As Long
    Dim j As Long
    Dim f As Long

    ReDim r(0 To dvLen - 1) As Byte

    For i = 0 To dLen - 1
        f = (CLng(data(i)) Xor CLng(r(0))) And &HFF

        For j = 0 To dvLen - 2
            r(j) = r(j + 1)
        Next j

        r(dvLen - 1) = 0

        If f <> 0 Then
            For j = 0 To dvLen - 1
                r(j) = r(j) Xor CByte(GFMul(CLng(dv(j)), f))
            Next j
        End If
    Next i

    RSRem = r
End Function

Private Function ToBytes(ByVal s As String) As Byte()
    InitEnc

    Dim cp As Long
    Dim buf() As Byte
    Dim nBytes As Long

    Dim pSrc As LongPtr
#Else
    Dim pSrc As Long

' ===== QR漢字モード判定・変換 =====
Private Function IsQRKanjiPair(ByVal b1 As Byte, ByVal b2 As Byte) As Boolean
    Dim sj As Long

    sj = CLng(b1) * 256 + CLng(b2)

    IsQRKanjiPair = _
        ((sj >= &H8140 And sj <= &H9FFC) Or _
         (sj >= &HE040 And sj <= &HEBBF))
End Function

Private Function CanEncodeQRKanji(u() As Byte, ByVal uLen As Long) As Boolean
    Dim i As Long

    If uLen <= 0 Then
        CanEncodeQRKanji = False
        Exit Function
    End If

    If (uLen Mod 2) <> 0 Then
        CanEncodeQRKanji = False
        Exit Function
    End If

    For i = 0 To uLen - 1 Step 2
        If Not IsQRKanjiPair(u(i), u(i + 1)) Then
            CanEncodeQRKanji = False
            Exit Function
        End If
    Next i

    CanEncodeQRKanji = True
End Function

Private Function QRKanjiCode(ByVal b1 As Byte, ByVal b2 As Byte) As Long
    Dim sj As Long
    Dim subVal As Long

    sj = CLng(b1) * 256 + CLng(b2)

    If sj >= &H8140 And sj <= &H9FFC Then
        subVal = sj - &H8140
    Else
        subVal = sj - &HC140
    End If

    QRKanjiCode = ((subVal \ 256) * &HC0) + (subVal And &HFF)
End Function

Private Sub AddBits(bits() As Long, ByRef bLen As Long, ByVal value As Long, ByVal count As Long)
    Dim i As Long

    For i = count - 1 To 0 Step -1
        bits(bLen) = (value \ PowOf2(i)) And 1
        bLen = bLen + 1
    Next i
End Sub

Private Function ECCPerBlock(ByVal v As Long) As Long
    Dim t As Variant

    ECCPerBlock = CLng(t(v))
End Function

Private Function NumECBlocks(ByVal v As Long) As Long
    Dim t As Variant

    t = Array(-1, 1, 1, 1, 2, 2, 4, 4, 4, 5, 5, 5, 8, 9, 9, 10, 10, 11, 13, 14, 16, 17, 17, 18, 20, 21, 23, 25, 26, 28, 29, 31, 33, 35, 37, 38, 40, 43, 45, 47, 49)

    NumECBlocks = CLng(t(v))
End Function

Private Function RawDataMods(ByVal v As Long) As Long
    Dim r As Long
    Dim n As Long

    r = (16 * v + 128) * v + 64

    If v >= 2 Then
        n = (v \ 7) + 2
        r = r - ((25 * n - 10) * n - 55)

        If v >= 7 Then
            r = r - 36
        End If
    End If

    RawDataMods = r
End Function

Private Function NumDataCW(ByVal v As Long) As Long
    NumDataCW = RawDataMods(v) \ 8 - ECCPerBlock(v) * NumECBlocks(v)
End Function

Private Function DetVer(ByVal dLen As Long) As Long
    Dim v As Long
    Dim cc As Long
    Dim eciBits As Long
    Dim dataBits As Long

    For v = 1 To 40
        If gQRKanjiMode Then
            If v <= 9 Then
                cc = 8
            ElseIf v <= 26 Then
                cc = 10
            Else
                cc = 12
            End If

            eciBits = 0
            dataBits = (dLen \ 2) * 13
        Else
            If v <= 9 Then
                cc = 8
            Else
                cc = 16
            End If

            If UseECI() Then
                eciBits = 12
            Else
                eciBits = 0
            End If

            dataBits = dLen * 8
        End If

        If eciBits + 4 + cc + dataBits <= NumDataCW(v) * 8 Then
            DetVer = v
            Exit Function
        End If
    Next v

    DetVer = -1
End Function

Private Function BuildDCW(u() As Byte, ByVal uLen As Long, ByVal v As Long) As Byte()
    Dim total As Long
    Dim bits() As Long
    Dim bLen As Long
    Dim i As Long
    Dim j As Long
    Dim cc As Long
    Dim bv As Long

    total = NumDataCW(v)

    ReDim bits(0 To total * 8 - 1) As Long

    bLen = 0

    If gQRKanjiMode Then
        ' QR漢字モード
        AddBits bits, bLen, 8, 4

        If v <= 9 Then
            cc = 8
        ElseIf v <= 26 Then
            cc = 10
        Else
            cc = 12
        End If

        AddBits bits, bLen, uLen \ 2, cc

        For i = 0 To uLen - 1 Step 2
            AddBits bits, bLen, QRKanjiCode(u(i), u(i + 1)), 13
        Next i
    Else
        ' バイトモード
        If UseECI() Then
            AddBits bits, bLen, 7, 4
            AddBits bits, bLen, 26, 8
        End If

        AddBits bits, bLen, 4, 4

        If v <= 9 Then
            cc = 8
        Else
            cc = 16
        End If

        AddBits bits, bLen, uLen, cc

        For i = 0 To uLen - 1
            AddBits bits, bLen, CLng(u(i)), 8
        Next i
    End If

    Dim tl As Long

    tl = total * 8 - bLen

    If tl > 4 Then
        tl = 4
    End If

    For i = 1 To tl
        bits(bLen) = 0
        bLen = bLen + 1
    Next i

    Do While (bLen Mod 8) <> 0
        bits(bLen) = 0
        bLen = bLen + 1
    Loop

    Dim pb As Long

    pb = &HEC

    Do While bLen < total * 8
        AddBits bits, bLen, pb, 8

        If pb = &HEC Then
            pb = &H11
        Else
            pb = &HEC
        End If
    Loop

    Dim cw() As Byte

    ReDim cw(0 To total - 1) As Byte

    For i = 0 To total - 1
        bv = 0

        For j = 0 To 7
            bv = bv * 2 + bits(i * 8 + j)
        Next j

        cw(i) = CByte(bv)
    Next i

    BuildDCW = cw
End Function

Private Function AddECC(dCW() As Byte, ByVal v As Long) As Byte()
    InitGF

    Dim nB As Long
    Dim bEC As Long
    Dim raw As Long
    Dim nSh As Long
    Dim shL As Long

    nB = NumECBlocks(v)
    bEC = ECCPerBlock(v)
    raw = RawDataMods(v) \ 8

    nSh = nB - (raw Mod nB)
    shL = raw \ nB

    Dim dv() As Byte
    Dim shD As Long
    Dim bData() As Byte
    Dim bDL() As Long
    Dim bECC() As Byte

    dv = RSDiv(bEC)
    shD = shL - bEC

    ReDim bData(0 To raw - 1)
    ReDim bDL(0 To nB - 1)
    ReDim bECC(0 To nB * bEC - 1)

    Dim dOff As Long
    Dim sK As Long
    Dim i As Long
    Dim j As Long
    Dim tDL As Long

    For i = 0 To nB - 1
        If i < nSh Then
            tDL = shD
        Else
            tDL = shD + 1
        End If

        bDL(i) = tDL

        Dim bb() As Byte

        ReDim bb(0 To tDL - 1) As Byte

        For j = 0 To tDL - 1
            bb(j) = dCW(sK)
            bData(dOff + j) = dCW(sK)
            sK = sK + 1
        Next j

        dOff = dOff + tDL

        Dim ec() As Byte

        ec = RSRem(bb, tDL, dv, bEC)

        For j = 0 To bEC - 1
            bECC(i * bEC + j) = ec(j)
        Next j
    Next i

    Dim res() As Byte
    Dim rI As Long
    Dim oA() As Long
    Dim cO As Long

    ReDim res(0 To raw - 1)
    ReDim oA(0 To nB - 1)

    For i = 0 To nB - 1
        oA(i) = cO
        cO = cO + bDL(i)
    Next i

    For j = 0 To shD
        For i = 0 To nB - 1
            If j = shD And i < nSh Then
                ' 短いブロックはここを飛ばす
            Else
                If j < bDL(i) Then
                    res(rI) = bData(oA(i) + j)
                    rI = rI + 1
                End If
            End If
        Next i
    Next j

    For j = 0 To bEC - 1
        For i = 0 To nB - 1
            res(rI) = bECC(i * bEC + j)
            rI = rI + 1
        Next i
    Next j

    AddECC = res
End Function

Private Sub SM(m() As Long, f() As Long, ByVal x As Long, ByVal y As Long, ByVal d As Long, ByVal sz As Long)
    If x >= 0 And x < sz And y >= 0 And y < sz Then
        m(y * sz + x) = d
        f(y * sz + x) = 1
    End If
End Sub

Private Sub DrawFP(m() As Long, f() As Long, ByVal cx As Long, ByVal cy As Long, ByVal sz As Long)
    Dim dx As Long
    Dim dy As Long
    Dim xx As Long
    Dim yy As Long
    Dim dt As Long
    Dim d As Long

    For dy = -4 To 4
        For dx = -4 To 4
            xx = cx + dx
            yy = cy + dy

            If xx >= 0 And xx < sz And yy >= 0 And yy < sz Then
                dt = Abs(dx)

                If Abs(dy) > dt Then
                    dt = Abs(dy)
                End If

                If dt <> 2 And dt <> 4 Then
                    d = 1
                Else
                    d = 0
                End If

                SM m, f, xx, yy, d, sz
            End If
        Next dx
    Next dy
End Sub

Private Sub DrawAP(m() As Long, f() As Long, ByVal cx As Long, ByVal cy As Long, ByVal sz As Long)
    Dim dx As Long
    Dim dy As Long
    Dim dt As Long
    Dim d As Long

    For dy = -2 To 2
        For dx = -2 To 2
            dt = Abs(dx)

            If Abs(dy) > dt Then
                dt = Abs(dy)
            End If

            If dt <> 1 Then
                d = 1
            Else
                d = 0
            End If

            SM m, f, cx + dx, cy + dy, d, sz
        Next dx
    Next dy
End Sub

Private Function GetAP(ByVal v As Long) As Long()
    Dim p() As Long
    Dim n As Long
    Dim st As Long
    Dim sz As Long
    Dim i As Long
    Dim pos As Long

    If v = 1 Then
        ReDim p(0 To 0)
        p(0) = -1
        GetAP = p
        Exit Function
    End If

    n = (v \ 7) + 2
    sz = v * 4 + 17

    st = ((sz - 13) * 2 + n - 2) \ ((n - 1) * 2)

    If (st Mod 2) = 1 Then
        st = st + 1
    End If

    ReDim p(0 To n - 1)

    p(0) = 6
    pos = sz - 7

    For i = n - 1 To 1 Step -1
        p(i) = pos
        pos = pos - st
    Next i

    GetAP = p
End Function

Private Sub DrawFunc(m() As Long, f() As Long, ByVal v As Long, ByVal sz As Long)
    Dim i As Long
    Dim j As Long
    Dim tv As Long

    For i = 0 To sz - 1
        If (i Mod 2) = 0 Then
            tv = 1
        Else
            tv = 0
        End If

        SM m, f, 6, i, tv, sz
        SM m, f, i, 6, tv, sz
    Next i

    DrawFP m, f, 3, 3, sz
    DrawFP m, f, sz - 4, 3, sz
    DrawFP m, f, 3, sz - 4, sz

    Dim ap() As Long

    ap = GetAP(v)

    If ap(0) <> -1 Then
        Dim nA As Long

        nA = UBound(ap) + 1

        For i = 0 To nA - 1
            For j = 0 To nA - 1
                If Not ((i = 0 And j = 0) Or (i = 0 And j = nA - 1) Or (i = nA - 1 And j = 0)) Then
                    DrawAP m, f, ap(i), ap(j), sz
                End If
            Next j
        Next i
    End If

    For i = 0 To 8
        If i < sz Then SM m, f, 8, i, 0, sz
        If i < sz Then SM m, f, i, 8, 0, sz
    Next i

    For i = 0 To 7
        SM m, f, sz - 1 - i, 8, 0, sz
        SM m, f, 8, sz - 1 - i, 0, sz
    Next i

    SM m, f, 8, sz - 8, 1, sz

    If v >= 7 Then
        Dim rm As Long
        Dim vb2 As Long
        Dim a As Long
        Dim b As Long
        Dim vc As Long

        rm = v

        For i = 0 To 11
            rm = (rm * 2) Xor (((rm \ 2048) And 1) * &H1F25)
        Next i

        vb2 = CLng(v) * 4096 Or rm

        For i = 0 To 17
            a = sz - 11 + (i Mod 3)
            b = i \ 3
            vc = (vb2 \ PowOf2(i)) And 1

            SM m, f, a, b, vc, sz
            SM m, f, b, a, vc, sz
        Next i
    End If
End Sub

Private Sub DrawFmt(m() As Long, f() As Long, ByVal mk As Long, ByVal sz As Long)
    Dim d2 As Long
    Dim rm As Long
    Dim i As Long
    Dim fb As Long

    d2 = 0 * 8 + mk
    rm = d2

    For i = 0 To 9
        rm = (rm * 2) Xor (((rm \ 512) And 1) * &H537)
    Next i

    fb = ((d2 * 1024) Or rm) Xor &H5412

    For i = 0 To 5
        SM m, f, 8, i, (fb \ PowOf2(i)) And 1, sz
    Next i

    SM m, f, 8, 7, (fb \ 64) And 1, sz
    SM m, f, 8, 8, (fb \ 128) And 1, sz
    SM m, f, 7, 8, (fb \ 256) And 1, sz

    For i = 9 To 14
        SM m, f, 14 - i, 8, (fb \ PowOf2(i)) And 1, sz
    Next i

    For i = 0 To 7
        SM m, f, sz - 1 - i, 8, (fb \ PowOf2(i)) And 1, sz
    Next i

    For i = 8 To 14
        SM m, f, 8, sz - 15 + i, (fb \ PowOf2(i)) And 1, sz
    Next i

    SM m, f, 8, sz - 8, 1, sz
End Sub

Private Sub DrawCW(m() As Long, f() As Long, cw() As Byte, ByVal cwL As Long, ByVal sz As Long)
    Dim bi As Long
    Dim r2 As Long
    Dim vt As Long
    Dim j2 As Long
    Dim x As Long
    Dim y As Long

    bi = 0
    r2 = sz - 1

    Do While r2 >= 1
        If r2 = 6 Then
            r2 = 5
        End If

        For vt = 0 To sz - 1
            For j2 = 0 To 1
                x = r2 - j2

                If ((r2 + 1) And 2) = 0 Then
                    y = sz - 1 - vt
                Else
                    y = vt
                End If

                If f(y * sz + x) = 0 And bi < cwL * 8 Then
                    m(y * sz + x) = (CLng(cw(bi \ 8)) \ PowOf2(7 - (bi Mod 8))) And 1
                    bi = bi + 1
                End If
            Next j2
        Next vt

        r2 = r2 - 2
    Loop
End Sub

Private Sub MaskApply(m() As Long, f() As Long, ByVal mk As Long, ByVal sz As Long)
    Dim x As Long
    Dim y As Long
    Dim inv As Boolean

    For y = 0 To sz - 1
        For x = 0 To sz - 1
            If f(y * sz + x) = 0 Then
                Select Case mk
                    Case 0
                        inv = ((x + y) Mod 2 = 0)
                    Case 1
                        inv = (y Mod 2 = 0)
                    Case 2
                        inv = (x Mod 3 = 0)
                    Case 3
                        inv = ((x + y) Mod 3 = 0)
                    Case 4
                        inv = ((x \ 3 + y \ 2) Mod 2 = 0)
                    Case 5
                        inv = ((x * y) Mod 2 + (x * y) Mod 3 = 0)
                    Case 6
                        inv = (((x * y) Mod 2 + (x * y) Mod 3) Mod 2 = 0)
                    Case 7
                        inv = (((x + y) Mod 2 + (x * y) Mod 3) Mod 2 = 0)
                End Select

                If inv Then
                    m(y * sz + x) = 1 - m(y * sz + x)
                End If
            End If
        Next x
    Next y
End Sub

Private Function Penalty(m() As Long, ByVal sz As Long) As Long
    Dim p As Long
    Dim x As Long
    Dim y As Long
    Dim c As Long
    Dim rc As Long
    Dim rl As Long

    For y = 0 To sz - 1
        rc = -1
        rl = 0

        For x = 0 To sz - 1
            c = m(y * sz + x)

            If c = rc Then
                rl = rl + 1

                If rl = 5 Then
                    p = p + 3
                End If

                If rl > 5 Then
                    p = p + 1
                End If
            Else
                rc = c
                rl = 1
            End If
        Next x
    Next y

    For x = 0 To sz - 1
        rc = -1
        rl = 0

        For y = 0 To sz - 1
            c = m(y * sz + x)

            If c = rc Then
                rl = rl + 1

                If rl = 5 Then
                    p = p + 3
                End If

                If rl > 5 Then
                    p = p + 1
                End If
            Else
                rc = c
                rl = 1
            End If
        Next y
    Next x

    For y = 0 To sz - 2
        For x = 0 To sz - 2
            c = m(y * sz + x)

            If c = m(y * sz + x + 1) And c = m((y + 1) * sz + x) And c = m((y + 1) * sz + x + 1) Then
                p = p + 3
            End If
        Next x
    Next y

    Dim dk As Long
    Dim tt As Long

    dk = 0
    tt = sz * sz

    For y = 0 To sz - 1
        For x = 0 To sz - 1
            If m(y * sz + x) = 1 Then
                dk = dk + 1
            End If
        Next x
    Next y

    Dim k As Long

    k = 0

    Do While Not ((50 - (k + 1) * 5) * tt <= dk * 100 And dk * 100 <= (50 + (k + 1) * 5) * tt)
        k = k + 1

        If k > 9 Then
            Exit Do
        End If
    Loop

    p = p + k * 10

    Penalty = p
End Function

Private Function GenQR(ByVal txt As String, m() As Long, sz As Long) As Boolean
    On Error GoTo Err1

    InitGF

    Dim u() As Byte
    Dim uL As Long
    Dim v As Long

    u = ToBytes(txt)
    uL = UBound(u) + 1

    ' Shift-JISを選んだ場合で、全体がQR漢字モードにできる文字なら漢字モードにする
    gQRKanjiMode = (Not gUseUTF8) And CanEncodeQRKanji(u, uL)

    v = DetVer(uL)

    If v < 1 Then
        GoTo Err1
    End If

    sz = v * 4 + 17

    ReDim m(0 To sz * sz - 1) As Long

    Dim f() As Long

    ReDim f(0 To sz * sz - 1) As Long

    Dim dCW() As Byte
    Dim aCW() As Byte

    dCW = BuildDCW(u, uL, v)
    aCW = AddECC(dCW, v)

    DrawFunc m, f, v, sz
    DrawCW m, f, aCW, UBound(aCW) + 1, sz

    Dim sv() As Long
    Dim ii As Long

    ReDim sv(0 To sz * sz - 1) As Long

    For ii = 0 To sz * sz - 1
        sv(ii) = m(ii)
    Next ii

    Dim bm As Long
    Dim bp As Long
    Dim cp As Long
    Dim mk As Long

    bp = 2000000000

    For mk = 0 To 7
        For ii = 0 To sz * sz - 1
            m(ii) = sv(ii)
        Next ii

        MaskApply m, f, mk, sz
        DrawFmt m, f, mk, sz

        cp = Penalty(m, sz)

        If cp < bp Then
            bp = cp
            bm = mk
        End If
    Next mk

    For ii = 0 To sz * sz - 1
        m(ii) = sv(ii)
    Next ii

    MaskApply m, f, bm, sz
    DrawFmt m, f, bm, sz

    GenQR = True
    Exit Function

Err1:
    GenQR = False
End Function

    Put #fn, , CByte((v \ 256) And 255)
    Put #fn, , CByte((v \ 65536) And 255)
    Put #fn, , CByte((v \ 16777216) And 255)
End Sub

Private Sub SaveBMP(m() As Long, ByVal qsz As Long, ByVal fp As String)
    Dim isz As Long
    Dim rb As Long
    Dim pb As Long
    Dim fn As Long

    isz = (qsz + MARGIN_MOD * 2) * CELL_PX
    rb = ((isz * 3 + 3) \ 4) * 4
    pb = rb - isz * 3

    fn = FreeFile

    Open fp For Binary As #fn

    Put #fn, , CByte(66)
    Put #fn, , CByte(77)

    PL fn, 54 + rb * isz
    PL fn, 0
    PL fn, 54
    PL fn, 40
    PL fn, isz
    PL fn, isz

    Put #fn, , CByte(1)
    Put #fn, , CByte(0)
    Put #fn, , CByte(24)
    Put #fn, , CByte(0)

    PL fn, 0
    PL fn, rb * isz
    PL fn, 2835
    PL fn, 2835
    PL fn, 0
    PL fn, 0

    Dim rw() As Byte
    Dim py As Long
    Dim px As Long
    Dim mx As Long
    Dim my As Long
    Dim bi As Long
    Dim pi As Long
    Dim idx As Long

    ReDim rw(0 To rb - 1) As Byte

    For py = isz - 1 To 0 Step -1
        bi = 0

        For px = 0 To isz - 1
            mx = (px \ CELL_PX) - MARGIN_MOD
            my = (py \ CELL_PX) - MARGIN_MOD

            If mx >= 0 And mx < qsz And my >= 0 And my < qsz Then
                idx = my * qsz + mx

                If idx >= 0 And idx <= UBound(m) Then
                    If m(idx) = 1 Then
                        rw(bi) = 0
                        rw(bi + 1) = 0
                        rw(bi + 2) = 0
                    Else
                        rw(bi) = 255
                        rw(bi + 1) = 255
                        rw(bi + 2) = 255
                    End If
                Else
                    rw(bi) = 255
                    rw(bi + 1) = 255
                    rw(bi + 2) = 255
                End If
            Else
                rw(bi) = 255
                rw(bi + 1) = 255
                rw(bi + 2) = 255
            End If

            bi = bi + 3
        Next px

        For pi = 0 To pb - 1
            rw(bi + pi) = 0
        Next pi

        Put #fn, , rw
    Next py

    Close #fn
End Sub

Private Function PlaceQR(ws As Worksheet, txt As String, tc As Range) As Boolean
    Dim m() As Long
    Dim sz As Long
    Dim tp As String
    Dim sn As String
    Dim side As Double

    On Error GoTo E1

    If Not GenQR(txt, m, sz) Then
        PlaceQR = False
        Exit Function
    End If

    sn = "QR_" & tc.Row

    On Error Resume Next
    ws.Shapes(sn).Delete
    On Error GoTo E1

    tp = Environ("TEMP") & "\qr_" & Format(Now, "yyyymmddhhnnss") & "_" & tc.Row & ".bmp"

    SaveBMP m, sz, tp

    ' セルの大きさを設定
    tc.ColumnWidth = CELL_WIDTH
    tc.RowHeight = CELL_HEIGHT

    ' セル枠いっぱい（短い辺に合わせた正方形）にフィット
    side = WorksheetFunction.Min(tc.Width, tc.Height) - QR_PADDING * 2

    If side < 8 Then
        side = 8
    End If

    Dim sh As Shape

    Set sh = ws.Shapes.AddPicture(tp, msoFalse, msoTrue, 0, 0, side, side)

    sh.Name = sn

    ' セル内で中央寄せ
    sh.Left = tc.Left + (tc.Width - side) / 2
    sh.Top = tc.Top + (tc.Height - side) / 2
    sh.LockAspectRatio = msoTrue

    On Error Resume Next
    Kill tp
    On Error GoTo 0

    PlaceQR = True
    Exit Function

E1:
    PlaceQR = False
End Function

📐 なぜ「全角QRコード」が必要なのか

QRコード規格（JIS X 0510）自体は、バイトモード（モード指示子 0100）を使えば任意のバイト列を格納できます。つまり漢字・ひらがな・カタカナもQRコードに入れることは仕様上可能です。問題は「生成側」と「読み取り側」と「出力先アプリ」の全体で、同じ文字コードとして扱えるかどうかにあります。

バーコードコントロールの限界

Microsoft BarCode Control 16.0（ActiveX）は、Excelから挿入できる場合がありますが、Excel単体の標準機能ではなく、主にAccessに同梱されるコントロールです。Excelだけしか入っていないPCでは一覧に表示されない場合があります。Microsoft 365 / Office 2024では初期設定でActiveXが表示・作成できないことがありますが、トラストセンター設定で許可すると使える場合もあります。ただし、管理者設定によって完全にブロックされている職場PCでは作成・操作できません。また、利用できる環境でも内部で半角ASCII（0x20〜0x7E）しかエンコードしません。全角文字をLinkedCellに入れても無視されるか、文字化けしたQRが生成されます。これが記事⑨⑩で「半角データ限定」と注意していた理由です。

閉域環境の現実

インターネットに接続できない閉域ネットワーク・スタンドアロンPCでは、Pythonの qrcode ライブラリやnpmパッケージをインストールすることが難しく、外部DLLの利用や搬入もセキュリティポリシーで禁止されている場合がほとんどです。

ただし、すべての閉域環境で外部コードの利用が完全に不可能とは限りません。申請手続きを経て、管理者が承認した方法で .bas ファイルのインポートが許可されるケースもあります。その場合は、実装済みライブラリの活用が圧倒的に効率的です。VBA＋Windows標準APIでの自作は、それすらできない「完全閉域」環境を想定した最終手段です。本記事では、その最終手段として使える公開版コードを掲載します。

🔥 最重要 ─ 文字コードを「作る側」と「読む側」で必ず合わせる

全角QRの最大のハマりどころは文字コードの不一致による文字化けです。QRコードを作成するときに「Shift-JISで作るか、UTF-8で作るか」を決め、リーダー側もそれと同じ文字コードに設定しなければなりません。制限環境では追加ツールの導入可否も大きく効くため、実務上はShift-JIS運用を先に検討します。

🔥 鉄則：「QR生成時の文字コード」＝「リーダーの文字コード設定」＝「出力先アプリへ渡せる文字コード」
この3点を一致させてください。オプトエレクトロニクス機では、UTF-8運用はMOSTを使う漢字出力設定が必要です。Shift-JIS運用なら、漢字出力モード②とShift-JIS設定でExcelなどへ出力できます。

QRコード作成時の文字コード選択

VBAで全角QRを生成する際、文字列をバイト列に変換するステップがあります。このときUTF-8で変換するかShift-JISで変換するかによって、QRコードの中身が根本的に変わります。たとえば漢字「光」はShift-JISでは 8C F5（2バイト）、UTF-8では E5 85 89（3バイト）になります。まったく異なるバイト列がQRに格納されるため、読み取り側でも同じ文字コードを指定しなければ正しく復元できません。

ℹ️ コード内部の補足：ECI ECI（Extended Channel Interpretation）は、QRコード内に「この後のデータはこの文字コードで読む」という情報を入れる仕組みです。このVBAコードにはUTF-8選択時だけECI番号26（UTF-8）を付ける処理があります。Shift-JIS選択時はECIを付けません。なお、ECIを付けてもOPI-3601側のUTF-8出力設定が不要になるわけではありません。UTF-8運用にはMOSTが必要なので、実運用の判断はコード直前の設定表を優先してください。

リーダー側の文字コード設定

QRコードリーダー（OPI-3601、ALX-3601、L-46X等）には、読み取ったバイト列をどの文字コードとしてPCに送るかを設定する項目があります。QRをShift-JISで作ったなら、リーダー側は漢字出力モード②＋Shift-JISにします。QRをUTF-8で作る場合は、リーダー側もUTF-8に合わせるだけでなく、MOSTを使える環境かを先に確認します。

出力先アプリケーションの文字コード

さらに、リーダーから送られたデータを受け取るアプリケーション側の文字コードも関係します。

📊 QRコード読み取りの信頼性と限界 ― 手入力ミスを大幅に減らせるが、検証は必要

QRコードは、手入力と比べて入力ミスの発生要因を大幅に減らせる強力な手段です。 手で文字を打つ作業を、機械的な読み取りに置き換えられるため、特に長いコード・住所・定型文・処理票データの転記では大きな効果があります。

ただし、QRコードの読み取り結果が常に100%保証されるわけではありません。 実運用上の信頼性は、QR生成実装、誤り訂正レベル、印刷品質、表示サイズ、リーダー機種、文字コード設定、出力先アプリケーションに依存します。 そのため、導入時には必ず読み取りテストと照合を行い、業務システム側でもバリデーションを併用する設計が必要です。

手入力 vs QRコード ― 減らせるミスの種類

入力方式起きやすいミス特徴確認の考え方
手入力 打ち間違い、桁抜け、変換ミス、読み間違い、コピペ位置ずれ 人の集中力・疲労・慣れに左右される。長い文字列や全角混在データほどミスが増えやすい 全件目視確認やダブルチェックが必要になりやすい
手入力＋ダブルチェック 見落とし、確認者側の判断ミス 精度は上がるが、工数が大きい。入力者と確認者の双方に負担がかかる 重要データでは有効だが、大量件数ではコストが高い
QRコード読み取り 印刷劣化、読み取り失敗、文字コード不一致、生成実装ミス、出力先アプリとの相性 手入力由来の打鍵ミスを大幅に減らせる。誤り訂正機能により、多少の汚れや欠けにも強い 導入時の読み取りテスト、自動バリデーション、必要に応じた抜き取り確認で運用する
✅ QRコードの強み：
QRコードにはリード・ソロモン誤り訂正が組み込まれており、印刷のかすれ・汚れ・一部欠損があっても、一定範囲内でデータを復元できます。 DENSO WAVEのQRコード解説でも、誤り訂正レベル L / M / Q / H が用意され、利用環境に応じて選択できることが説明されています。 ただし、これは「汚れや破損に対する復元能力」であり、実運用におけるエラー率そのものを保証するものではありません。

それでも確認が必要な理由

QRコードを使う目的は、「確認をゼロにすること」ではありません。 手入力による打ち間違い・桁抜け・変換ミスを減らし、確認作業を 全文目視確認から、自動バリデーションや抜き取り確認へ置き換えることです。

特に本記事のようにVBAでQRコードを生成する場合、リーダーの性能だけでなく、 VBA側のQR生成ロジックが正しいかも確認対象になります。 生成したQRを複数のリーダーやスマートフォンアプリで読み取り、元データと一致するかを照合してください。

リスク要因内容対策
QR生成実装の不具合 VBAで自作・移植したQR生成ロジックにバグがあると、規格上正しくないQRや、特定文字で壊れるQRが生成される可能性がある 既知のライブラリを優先。自作する場合は、英数字・漢字・記号・改行・長文など複数パターンで元データとの照合テストを行う
文字コードの不一致 生成時とリーダー設定が異なると、読み取り自体は成功しても文字化けしたデータが出力される UTF-8 / Shift-JIS のどちらで作ったかを明記し、リーダー側設定と必ず合わせる
印刷品質・表示品質 印刷のかすれ、にじみ、紙の折れ、画面反射、QRサイズ不足により読み取り失敗や不安定化が起きる 十分なサイズ、余白、コントラストを確保する。紙の場合は印刷テスト、画面表示の場合は輝度と反射を確認する
リーダー機種差 リーダーによって、全角出力、文字コード、長文QR、画面読み取り性能に差がある 本番で使うリーダー実機でテストする。スマホアプリだけの確認で済ませない
出力先アプリとの相性 Excel、Word、Webシステム、メモ帳、端末エミュレータで、Enter、Tab、文字コード、IMEの扱いが異なる 実際の入力先アプリでテストする。メモ帳で成功しても業務システムで成功するとは限らない
運用上の読み飛ばし・重複読み取り 複数QRを連続で読む場合、1枚飛ばす、同じQRを2回読む、順番を間違える可能性がある 連番、件数、チェックサム、最終QRでの合計照合を入れる
🔥 結論：QRコードは手入力ミスを大幅に減らせますが、検証なしで100%保証されるものではありません。
特に全角QRやVBA生成QRでは、読み取り精度だけでなく、生成ロジック、文字コード、リーダー設定、出力先アプリとの相性まで含めて確認する必要があります。 全角文字を含むQRを使う場合は、十分に注意したうえで、QRが作成されることだけでなく、実際の読み取り先に文字列が正しく出力されるかを確認してから使用してください。 ただし、これはQR方式の価値を下げるものではありません。 手入力の大量ミスを減らし、確認作業を「全文目視」から「自動チェック・異常検出・抜き取り確認」へ移せることが、QR入力の大きなメリットです。

確認が必要でも、QR化する意味はある

「確認が必要ならQR化する意味がない」と感じるかもしれませんが、実務上は逆です。 QR化の目的は、確認作業をなくすことではなく、手入力ミスを減らし、確認対象を絞ることです。

手入力では、入力された文字列そのものを人間が確認する必要があります。 一方、QR入力では、元データと読み取り結果の一致確認、チェックディジット、件数照合、マスタ存在チェックなど、機械的な確認に置き換えられます。 これにより、確認工数を減らしながら、手入力より安定した運用にできます。

業務システムでの二重チェック設計 ― QR＋バリデーションで万全にする

QRコードを実務で安全に使うには、「QRなら確認不要」と考えるのではなく、 QR入力＋業務システム側のバリデーションで異常を検出する設計にします。 これにより、人間が全文を目視確認する作業を減らし、異常値・桁違い・読み飛ばし・文字コード不一致を機械的に発見できます。

チェック方法具体例効果
クロスフィールドチェック 社員番号と生年月日の組み合わせをマスタと照合。QRから読み取った社員番号「12345」と生年月日「1990-05-15」が既存マスタの組み合わせと一致するか検証 社員番号が1桁ずれていても、生年月日との組み合わせが一致しなければエラーとして検出できる

✅ 実務での推奨パターン： たとえば顧客マスタ移行であれば、QRデータに「顧客コード＋氏名＋生年月日＋住所＋電話番号」を格納し、業務システム側で「顧客コードと氏名の組み合わせ」「電話番号の桁数」「郵便番号のフォーマット」をバリデーションする設計にします。 QR入力で手入力ミスを減らし、バリデーションで異常値や不整合を検出することで、安全性を高められます。

紙媒体のセキュリティ管理

QRコードを印刷した紙は、データを物理的に持ち出す媒体です。顧客情報・個人情報を含むQRコードを紙に印刷した場合、その紙は機密文書と同等の扱いが必要です。読み取り完了後は速やかにシュレッダーで廃棄することを徹底してください。

タブレット表示による紙レス運用 ― コスト削減と機密性の両立

QRコードは必ずしも紙に印刷する必要はありません。タブレットやPCの画面にQRコードを表示し、リーダーで画面を直接読み取る運用も可能です。この方法には以下のメリットがあります。

比較項目紙に印刷タブレット/PC画面表示
印刷コスト用紙代＋インク/トナー代がかかる。QRコードは黒モジュールの面積が大きいため、通常の文書より黒インク消費量が多く、大量印刷ではコストが目立つ不要（電気代のみ）
廃棄の手間シュレッダー廃棄が必須表示を閉じるだけ。物理的な残存リスクなし
機密性紙が残る限り情報漏洩リスクあり画面を閉じれば消える。使い捨て感覚で運用可能
読み取り品質印刷品質に依存（かすれ・にじみリスク）画面解像度が高く安定した読み取りが可能。ただし画面の反射・グレアに注意
環境・条件プリンタがあれば可能タブレットやサブモニターが必要。画面輝度を十分に上げる必要あり
✅ タブレット表示の実践方法：公開版コードでQRコードをBMP画像として生成し、Excelセルへ配置 → Excelブックをタブレット（iPadやSurface等）で開く → リーダーで画面上のQRを読み取り。紙を一枚も使わずにデータ転送が完了します。定型文入力用のQRカードも、ラミネート加工の代わりにタブレット表示にすれば制作コストがゼロになります。
⚠️ 画面読み取り時の注意：画面の反射（グレア）やブルーライトフィルムの影響で読み取りにくくなる場合があります。画面輝度を最大付近に設定し、リーダーを画面に対して正面から当てると安定します。また、画面のスリープ設定を長めにしておくと、連続読み取り時に画面が消えるトラブルを防げます。

紙印刷を行う場合のコスト意識

紙に印刷する場合、QRコードは黒モジュール（ドット）の面積比率が高いため、通常のテキスト文書と比べてインク・トナーの消費量が多くなります。特にレーザープリンタのトナーやインクジェットの黒インクは、A4用紙にQRコードを6枚並べて印刷するとかなりの消費になります。大量のデータ移行（数百枚規模）では、印刷コストも事前に見積もっておいてください。可能であればタブレット表示との併用を検討することで、コストを大幅に抑えられます。

QRリーダー連携の運用注意

QRリーダーは多くの場合、PCにはキーボード入力として認識されます。Tab、Enter、ファンクションキー、待機時間を組み合わせると登録・更新・削除などの操作まで進められるため、実務投入前の検証手順を必ず固定してください。

  • 本番データではなく、架空データと検証用画面で読み取り順、セル移動、登録操作を確認する。
  • 個人情報、認証情報、機密コードをQRにそのまま埋め込まない。必要な場合は最小限のIDに置き換える。
  • 設定バーコード、初期化手順、復旧手順を保存し、誰が変更したかを記録する。
  • キー変換や待機時間を変更した後は、Excel以外の画面がアクティブな状態で誤送信されないか確認する。
